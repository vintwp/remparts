Процедура ЗаписьТекстовогоФайла(ПутьКФайлу,ДанныеДляЗаписи)

КонецПроцедуры

Функция СтрокаJson(Параметр, Значение, ТипЗн = 0, ПоследняяСтрока = 0) // ТипЗп  0 - обьект или строка, 1 - число, 3 - массив; Последняя строка 0 - ставить зпт в конце, 1 - ставить
	ПараметрДляДобавления = Симв(34) + Параметр + Симв(34);
	ЗначениеДляДобавления = "";
	
	Если ТипЗн = 1 Тогда
		ЗначениеДляДобавления = Число(Значение);
	ИначеЕсли ТипЗн = 2 Тогда
		ЗначениеДляДобавления = "[" + Значение + "]";
	Иначе
		ЗначениеДляДобавления = Симв(34) + СокрЛП(Значение) + Симв(34);
	КонецЕсли;
	
	Если ПоследняяСтрока = 1 Тогда
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления);
	Иначе
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления + ",");
	КонецЕсли;
	              
КонецФункции


Функция УдалитьПоследнююЗапятую(стр)
	ДлинаСтроки = СтрДлина(стр) - 1; // без учета последней запятой
	СтрокаФормат = Лев(стр, ДлинаСтроки);
	
	Возврат(СтрокаФормат);
КонецФункции

Процедура СформироватьПрайс()
	СпрТмц = СоздатьОбъект("Справочник.ТМЦ");
	ПрайсJson = "";
	Строки = 1;
	
	СпрТмц.ПорядокКодов();
	
	ТестИд = "00011439";
	
	
	Если СпрТмц.ВыбратьЭлементы() = 1 Тогда
		Пока ((СпрТмц.ПолучитьЭлемент() = 1)) Цикл      //И (Строки < 10000)
			ТекущийЭлемент = СпрТмц.ТекущийЭлемент();
			Если ((ТекущийЭлемент.этоГруппа() = 1) ИЛИ (ТекущийЭлемент.ПометкаУдаления() = 1)) Тогда   // ИЛИ (ТекущийЭлемент.Код <> ТестИд)
				Продолжить;
			Иначе
				
				// НАЧАЛО собираем остатки ТМЦ
				СпрТмцДляТекущегоЭлемента = СоздатьОбъект("Справочник.ТМЦ");
				СпрТмцДляТекущегоЭлемента.НайтиПоКоду(ТекущийЭлемент.Код);
								
				Склад = СоздатьОбъект("Справочник.Склады");
				Склад.ВыборГруппы(1);
				Склад.ВыбратьЭлементы();
				
				Остаток = 0;
								
				Пока Склад.ПолучитьЭлемент() = 1 Цикл
					ТекущийСклад = Склад.ТекущийЭлемент();
					
					Если (ТекущийСклад.Код = "1") Тогда                  
						Остаток = Остаток + Регистр.Остатки.СводныйОстаток(ТекущийСклад, СпрТмцДляТекущегоЭлемента,, "Кво");
					КонецЕсли;
				КонецЦикла;
				
				
				// КОНЕЦ собираем остатки ТМЦ				
				        
				СтрокаJson = "{" 
					+ СтрокаJson("id", ТекущийЭлемент.Код,,0)
					+ СтрокаJson("name", ТекущийЭлемент.Наименование,,0)
					+ СтрокаJson("price", ТекущийЭлемент.ЦенаРозничная.Получить(ТекущаяДата()),1,0)
					+ СтрокаJson("priceWholesaleBasic", ТекущийЭлемент.Цена3Кат.Получить(ТекущаяДата()),1,0)
					+ СтрокаJson("priceWholesaleStandard", ТекущийЭлемент.Цена2Кат.Получить(ТекущаяДата()),1,0)
					+ СтрокаJson("priceWholesaleTop", ТекущийЭлемент.Цена1Кат.Получить(ТекущаяДата()),1,0)
					+ СтрокаJson("id", ТекущийЭлемент.АфмИд,,0)
					+ СтрокаJson("stock", Остаток,1,1)
					+ "}";
					
				ПрайсJson = ПрайсJson  + СтрокаJson + ",";             
				
				Строки = Строки + 1;
				
				Состояние("Сформировать прайс - обработано строк " + Строки);
			КонецЕсли;

		КонецЦикла;

		ПрайсJson = "[" + УдалитьПоследнююЗапятую(СокрЛП(ПрайсJson)) + "]";
		
		ЗаписьТекстовогоФайла(РабочаяПапка + "price.json",ПрайсJson);
		
		Сообщить("INFO: Прайс сформирован. Всего обработано строк - " + Строки);
		                    
	Иначе
		Сообщить("ERR: Справочник пуст!");
	КонецЕсли;
	              
КонецПроцедуры

Процедура ПолучитьКурс()
	СпрВалюты = СоздатьОбъект("Справочник.Валюты");
	Если СпрВалюты.НайтиПоКоду("001") = 1 Тогда    // 001 - ид гривны
		ЗначениеКурсаВалюты = СпрВалюты.Курс.Получить();

		ЗаписьТекстовогоФайла(РабочаяПапка + "currency.json",КурсJson);
		
	Иначе
		Сообщить("ERR: Элемент валюты не найден");
	КонецЕсли;
	
КонецПроцедуры

Процедура ПолучитьКонтрагентов()
	СпрКонтрагенты = СоздатьОбъект("Справочник.Контрагенты");
	СпрКонтрагенты.ВыбратьЭлементы();
	
	КонтрагентыJson = "";            
	                                  
	// Контрагенты групп, которых не должно быть в выгрузке - Учредители, Поставщики и Прочее...
	СпрКонтрагентыДляПоставщиков = СоздатьОбъект("Справочник.Контрагенты");
	СпрКонтрагентыДляПоставщиков.НайтиПоКоду("6"); // Код 6 - поставщики
	
	СпрКонтрагентыДляУчредителей = СоздатьОбъект("Справочник.Контрагенты");
	СпрКонтрагентыДляУчредителей.НайтиПоКоду("5"); // Код 5 - учредители
	
	ПоставщикиГруппа = СпрКонтрагентыДляПоставщиков.ТекущийЭлемент();
	УчредителиГруппа = СпрКонтрагентыДляУчредителей.ТекущийЭлемент();
	
	// ---------- 
	
	// приход денег от клиента - значок доллара
	// 
	
	Пока СпрКонтрагенты.ПолучитьЭлемент() = 1 Цикл
	    Контрагент = СпрКонтрагенты.ТекущийЭлемент();
		КонтрагентJson = СоздатьОбъект("Текст");
		Если Контрагент.ЭтоГруппа() = 1 Тогда
			Продолжить;
		Иначе
			Если ((Контрагент.Родитель <> ПоставщикиГруппа) И (Контрагент.Родитель <> УчредителиГруппа)) Тогда // И (Контрагент.Код = "16")
				
				Баланс = ПолучитьБалансКонтрагента(Контрагент);
				ПрихНаклКонтрагента = ПолучитьПрихНаклКонтрагента(Контрагент);
				КатегорияЦенКонтрагента = Контрагент.КатегорияЦенКонтрагента;
				  
				КатегорияЦенДляБдСайта = "RETAIL";
				
				Если КатегорияЦенКонтрагента = "Цена 1 категории" Тогда
					 КатегорияЦенДляБдСайта = "WHOLESALE_TOP";
				ИначеЕсли КатегорияЦенКонтрагента = "Цена 2 категории" Тогда
					 КатегорияЦенДляБдСайта = "WHOLESALE_STANDARD";
				ИначеЕсли КатегорияЦенКонтрагента = "Цена 3 категории" Тогда
					 КатегорияЦенДляБдСайта = "WHOLESALE_BASIC";
				ИначеЕсли КатегорияЦенКонтрагента = "Оптовая" Тогда
					 КатегорияЦенДляБдСайта = "WHOLESALE_BASIC";
				КонецЕсли;

				
				КонтрагентJson = "{"
					+ СтрокаJson("id1c", Контрагент.Код,, 0)
					+ СтрокаJson("name1с", Контрагент.ПолнНаименование,, 0)
					+ СтрокаJson("customerPriceTier", КатегорияЦенДляБдСайта,, 0)
					+ СтрокаJson("balance", Баланс,1, 0)
					+ СтрокаJson("payments", ПрихНаклКонтрагента, 2, 1)
					+ "}"; 

				КонтрагентыJson = КонтрагентыJson + КонтрагентJson + ",";			

			КонецЕсли;
		КонецЕсли;
	КонецЦикла;
	
	
	
	КонтрагентыJson = УдалитьПоследнююЗапятую(СокрЛП(КонтрагентыJson));
	
	КонтрагентыJson = "[" + СимволПереноса + КонтрагентыJson + СимволПереноса + "]";
	
	ЗаписьТекстовогоФайла(РабочаяПапка + "clients.json",КонтрагентыJson);
