Функция ПреобразоватьСтрокуВУТФ8(Стр)
    Длина=СтрДлина(Стр);
    Итог="";
    Для Н=1 По Длина Цикл
        Знак=Сред(Стр,Н,1);
        Код=КодСимв(Знак);
        Если Код<128 Тогда
            Итог=Итог+Знак;
        Иначе
            Если (Код>=КодСимв("А"))И(Код<=КодСимв("п")) Тогда
                Итог=Итог+Симв(208)+Симв(144+Код-КодСимв("А"));
            ИначеЕсли (Код>=КодСимв("р"))И(Код<=КодСимв("я")) Тогда
                Итог=Итог+Симв(209)+Симв(128+Код-КодСимв("р"));
            ИначеЕсли (Знак="ё") Тогда
                Итог=Итог+Симв(209)+Симв(145);
            ИначеЕсли (Знак="Ё") Тогда
                Итог=Итог+Симв(208)+Симв(129);
			ИначеЕсли (Знак = "і") Тогда
                Итог = Итог + Симв(209) + Симв(150);
            ИначеЕсли (Знак = "І") Тогда
                Итог = Итог + Симв(208) + Симв(134);
            ИначеЕсли (Знак = "ї") Тогда
                Итог = Итог + Симв(209) + Симв(151);
            ИначеЕсли (Знак = "Ї") Тогда
                Итог = Итог + Симв(208) + Симв(135);
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
    Возврат Итог;

Функция СтрокаJson(Параметр, Значение, ТипЗн = 0, ПоследняяСтрока = 0) // ТипЗп  0 - обьект или строка, 1 - число, 2 - массив; Последняя строка 0 - ставить зпт в конце, 1 - ставить
	ПараметрДляДобавления = Симв(34) + Параметр + Симв(34);
	ЗначениеДляДобавления = "";
	
	Если ТипЗн = 1 Тогда
		ЗначениеДляДобавления = Число(Значение);
	ИначеЕсли ТипЗн = 2 Тогда
		ЗначениеДляДобавления = "[" + Значение + "]";
	Иначе
		ФорматЗначение = ОчиститьТекстОтПереносов(Значение);
		
		Если Найти(ФорматЗначение, Симв(92)) <> 0 Тогда
			ЭкранированныйОбратныйСлеш = Симв(92) + Симв(92); //  \ -> \\ 
			ФорматЗначение = СтрЗаменить(Значение,Симв(92), ЭкранированныйОбратныйСлеш); // экранируем обратный слеш для json			
		КонецЕсли;
		
		
		Если Найти(ФорматЗначение, Симв(34)) <> 0 Тогда
			ЭкранированнаяКавычка = Симв(92) + Симв(34); //  \" 
			ФорматЗначение = СтрЗаменить(Значение,Симв(34), ЭкранированнаяКавычка); // экранируем кавычки для json			
		КонецЕсли;
		                                                         
		ЗначениеДляДобавления = Симв(34) + ФорматЗначение + Симв(34);
	КонецЕсли;
	
	Если ПоследняяСтрока = 1 Тогда
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления);
	Иначе
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления + ",");
	КонецЕсли;

Функция ОтправитьPOSTЗапрос(URL, ТекстJSON)
   
	ЗагрузитьВнешнююКомпоненту("OleExSup.dll");
	OleExSup = СоздатьОбъект("OLEExSup");
	
	ТелоОтветаНаЗапрос = СоздатьОбъект("СписокЗначений");
	ТелоОтветаНаЗапрос.ВставитьЗначение(1, 200);
	ТелоОтветаНаЗапрос.ДобавитьЗначение(2, "");

    Попытка                                                                                              
        HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
	Исключение
		ТелоОтветаНаЗапрос.ВставитьЗначение(1, 500);

        Возврат ТелоОтветаНаЗапрос;
    КонецПопытки;

    // Открываем соединение: ("Метод", "URL", Асинхронный = 0)
    HTTP.Open("POST", URL, 0);

    // Устанавливаем заголовки. Content-Type обязателен для JSON.
    // Указание кодировки UTF-8.
    HTTP.SetRequestHeader("Content-Type", "application/json; charset=utf-8");

    // Отправляем тело запроса (нашу строку JSON)
    Попытка
        // HTTP.Send(ТекстJSON);
		OleExSup.InvokeOLEMethod(HTTP, "Send", ТекстJSON);

        Возврат ТелоОтветаНаЗапрос;
    КонецПопытки;


    // Проверяем результат
    Если HTTP.Status = 200 Тогда // 200 - это стандартный код успешного ответа "OK"
        // Запрос прошел успешно, возвращаем текст ответа от сервера
		ОтветССервера = HTTP.ResponseText;
		Сообщение = "SERVER OK: Статус: " + HTTP.Status + ", Сообщение от сервера: " + ПолучЗначениеПараметраСJson("message",ОтветССервера);

Процедура СформироватьПрайс()

	НачалоСинхронизации();
	
	СпрТмц = СоздатьОбъект("Справочник.ТМЦ");
	
	СпрСклады = СоздатьОбъект("Справочник.Склады");
	НеобходимыйСкладИд = "1"; // основной склад ид

	Склад = "";  // переменная склада
	
    Если СпрСклады.НайтиПоКоду(НеобходимыйСкладИд) = 1 Тогда
		Склад = СпрСклады.ТекущийЭлемент();
	Иначе
		Сообщить("ERR: Склад с кодом " + Склад + " не найден!");
        КонецСинхронизации();
        Возврат;
	КонецЕсли;
		

	ПрайсJson = "";
	Строки = 1;
	
	СпрТмц.ПорядокКодов();
	

	ГруппаДляИсключения1 = "00000148";	 // товары  с группы ХМусор
	ГруппаДляИсключения2 = "00006909";   // товары с группы Ремонт
	
	
	Если СпрТмц.ВыбратьЭлементы() = 1 Тогда
		Пока ((СпрТмц.ПолучитьЭлемент() = 1)) Цикл      //И (Строки < 10000)
			ТекущийЭлемент = СпрТмц.ТекущийЭлемент();
			Остаток = 0;
			
			// Если элемент это каталог (или группа), или товар, помеченный на удаление, - пропускаем
			
			Если ((ТекущийЭлемент.этоГруппа() = 1) ИЛИ (ТекущийЭлемент.ПометкаУдаления() = 1)) Тогда   // ИЛИ (ТекущийЭлемент.Код <> ТестИд)
				Продолжить;
			КонецЕсли;
			
			// Если элемент входит в группу, которую необходимо пропустить в прайсе
			
            Если ((ВходитВГруппу(ТекущийЭлемент, ГруппаДляИсключения1) = 1) ИЛИ (ВходитВГруппу(ТекущийЭлемент, ГруппаДляИсключения2) = 1)) Тогда

			КонецЕсли;

			СтрокаJson = "{"

				+ СтрокаJson("name", ТекущийЭлемент.Наименование,,0)
				+ СтрокаJson("price", ТекущийЭлемент.ЦенаРозничная.Получить(ТекущаяДата()),1,0)
				+ СтрокаJson("priceWholesaleBasic", ТекущийЭлемент.Цена3Кат.Получить(ТекущаяДата()),1,0)
				+ СтрокаJson("priceWholesaleStandard", ТекущийЭлемент.Цена2Кат.Получить(ТекущаяДата()),1,0)
				+ СтрокаJson("priceWholesaleTop", ТекущийЭлемент.Цена1Кат.Получить(ТекущаяДата()),1,0)
				+ СтрокаJson("afmId", ТекущийЭлемент.АфмИд,,0)
				+ СтрокаJson("categoryId", ТекущийЭлемент.КатИд,1,0)
				+ СтрокаJson("departmentId", ТекущийЭлемент.ДепИд,1,0)
				+ СтрокаJson("brandId", ТекущийЭлемент.БрендИд,1,0)
				+ СтрокаJson("qualityId", ТекущийЭлемент.КачИд,1,0)
				+ СтрокаJson("complianceId", ТекущийЭлемент.СовмИд,1,0)
				+ СтрокаJson("stock", Остаток,1,1)
				+ "}";
				
				
			ПрайсJson = ПрайсJson  + СтрокаJson + ",";
			ЗаписьТекстовогоФайла(РабочаяПапка + "price.json",ПрайсJson);
			
			Строки = Строки + 1;

		КонецЦикла;

	Иначе

Процедура ПолучитьКонтрагентов()

	ГруппаДляИсключенияПоставщики = "6";
	ГруппаДляИсключенияУчредители = "5";

	СпрКонтрагентыДляПоставщиков = СоздатьОбъект("Справочник.Контрагенты");
	СпрКонтрагентыДляПоставщиков.НайтиПоКоду("6"); // Код 6 - поставщики
	
	СпрКонтрагентыДляУчредителей = СоздатьОбъект("Справочник.Контрагенты");
	СпрКонтрагентыДляУчредителей.НайтиПоКоду("5"); // Код 5 - учредители
	
	ПоставщикиГруппа = СпрКонтрагентыДляПоставщиков.ТекущийЭлемент();
	УчредителиГруппа = СпрКонтрагентыДляУчредителей.ТекущийЭлемент();
	
	// ---------- 
	
	// приход денег от клиента - значок доллара
	// 
	
	Пока СпрКонтрагенты.ПолучитьЭлемент() = 1 Цикл
	    Контрагент = СпрКонтрагенты.ТекущийЭлемент();
		КонтрагентJson = СоздатьОбъект("Текст");
		
		Если Контрагент.ЭтоГруппа() = 1 Тогда
			Продолжить;
		КонецЕсли;
		
		Если ((ВходитВГруппу(Контрагент, ГруппаДляИсключенияПоставщики) = 1) ИЛИ (ВходитВГруппу(Контрагент, ГруппаДляИсключенияУчредители) = 1)) Тогда
        	Продолжить;
		КонецЕсли;
		
			// И (Контрагент.Код = "16")
				
		Баланс = ПолучитьБалансКонтрагента(Контрагент);
		ПрихКасОрдераКонтрагента = ПолучитьПрихКасОрдераКонтрагента(Контрагент);
		РасхНакладныеКонтрагента = ПолучитьРасхНаклКонтрагента(Контрагент);
		КатегорияЦенКонтрагента = Контрагент.КатегорияЦенКонтрагента;
		  
		КатегорияЦенДляБдСайта = "RETAIL";

			 КатегорияЦенДляБдСайта = "WHOLESALE_TOP";
		ИначеЕсли КатегорияЦенКонтрагента = "Цена 2 категории" Тогда
			 КатегорияЦенДляБдСайта = "WHOLESALE_STANDARD";
		ИначеЕсли КатегорияЦенКонтрагента = "Цена 3 категории" Тогда
			 КатегорияЦенДляБдСайта = "WHOLESALE_BASIC";
		ИначеЕсли КатегорияЦенКонтрагента = "Оптовая" Тогда

		КонецЕсли;

			+ СтрокаJson("id1c", Контрагент.Код,, 0)
			+ СтрокаJson("name1c", Контрагент.ПолнНаименование,, 0)
