Перем Excel;

Процедура ПодключитьExcel()
	Попытка
		// Подключение к MS Excel
    	Excel = СоздатьОбъект("Excel.Application");
		Сообщить("INFO: MS Excel успешно подключен");
	Исключение
		// Если MS Excel не установлен на ПК
		Предупреждение("ERR: " + ОписаниеОшибки() + "; MS Excel не найден");
    	Excel = 0;
   		Возврат;
    КонецПопытки;
КонецПроцедуры

Процедура ОткрытьФайлВExcel(ПутьКФайлу)
	Попытка	
		Excel.Workbooks.Close();
		Excel.Workbooks.Open("D:" + ПутьКФайлу);
		Книга = Excel.ActiveWorkbook;
		Сообщить("Файл успешно загружен по пути - D:\" + ПутьКФайлу);
	Исключение
		Сообщить("Ошибка при открытии файла Excel: " + ОписаниеОшибки());
		Возврат;
	КонецПопытки;
КонецПроцедуры  

Процедура ОткрытьФайл()
	ИмяВыбрФайла = "";
	ИмяПути=КаталогИБ()+"ProgFile\";
	Если ФС.ВыбратьФайл(0, ИмяВыбрФайла, ИмяПути,"Выберите импортируемый файл", "Все файлы Excel (*.xls)|*.xls", , ) = 1 Тогда
		ПутьИмпортируемыйФайл=СРЕД((ИмяПути + ИмяВыбрФайла),3,100);
		плИмпортируемыйФайл = ПутьИмпортируемыйФайл;
		Форма.кнИмпорт.Доступность(1);
		ОткрытьФайлВExcel(ПутьИмпортируемыйФайл);
		Возврат;
	КонецЕсли;
КонецПроцедуры
	

	                                                        
// Проверить файл на корректность (Всего 8 колонок). Колонки Код, Наименование, Цены, КатИд, ДепИд - не могут быть пустыми!
// Импортируемый файл должен иметь колонки Код|Наименование|ЦенаОптовая|ЦенаРозничная|Цена 1 категории|Цена 2 категории|Цена 3 категории|Акционный

// Прим. Последняя колонка, содержит значение 0 или 1. ЦенаОптовая|ЦенаРозничная|Цена 1 категории|Цена 2 категории|Цена 3 категории|Акционный - числовой формат; Имя, Код, АфмИд - текстовый

Функция ПроверитьФайлНаКорректность()
	КолСтрок = Excel.Cells.CurrentRegion.Rows.Count;
	СтрокиСОшибками = "";
	
	Сообщить("Импорт в процессе");
	
	Для НомерСтроки=2 по КолСтрок Цикл
		// Получаем данные из ячеек
 		Код = СокрЛП(Excel.Cells(НомерСтроки, 1).Value);
 		Наименование = СокрЛП(Excel.Cells(НомерСтроки, 2).Value);
 		ЦенаОптовая = Число(Excel.Cells(НомерСтроки, 3).Value);
	 	ЦенаРозничная = Число(Excel.Cells(НомерСтроки, 4).Value);
 		ЦенаПервойКатегории = Число(Excel.Cells(НомерСтроки, 5).Value);
 		ЦенаВторойКатегории = Число(Excel.Cells(НомерСтроки, 6).Value);
 		ЦенаТретьейКатегории = Число(Excel.Cells(НомерСтроки, 7).Value);
 		Акционный = Число(Excel.Cells(НомерСтроки, 8).Value);
		АфмИд=Число(Excel.Cells(НомерСтроки, 9).Value);
		КатИд=Число(Excel.Cells(НомерСтроки, 10).Value);
		ДепИд=Число(Excel.Cells(НомерСтроки, 11).Value);
		БрендИд=Число(Excel.Cells(НомерСтроки, 12).Value);
		КачИд=Число(Excel.Cells(НомерСтроки, 13).Value);
		СовмИд=Число(Excel.Cells(НомерСтроки, 14).Value);
		
		Состояние("Проверка строки " + НомерСтроки + " из строк " + КолСтрок);
		
 		Попытка
 			пКод = Строка(СокрЛП(Код));
 			пНаименование = Строка(СокрЛП(Наименование));
 			пЦенаОптовая = Число(СокрЛП(ЦенаОптовая));
 			пЦенаРозничная = Число(СокрЛП(ЦенаРозничная));
 			пЦенаПервойКатегории = Число(СокрЛП(ЦенаПервойКатегории));
 			пЦенаВторойКатегории = Число(СокрЛП(ЦенаВторойКатегории));
 			пЦенаТретьейКатегории = Число(СокрЛП(ЦенаТретьейКатегории));
 			пАкционный = Число(СокрЛП(Акционный));
 			пАфмИд = Число(СокрЛП(АфмИд));
			пКатИд = Число(СокрЛП(КатИд));
			пДепИд = Число(СокрЛП(ДепИд));
			пБрендИд = Число(СокрЛП(БрендИд));
			пКачИд = Число(СокрЛП(КачИд));
			пСовмИд = Число(СокрЛП(СовмИд));
			
			
			Если (ПустоеЗначение(Код)=1) ИЛИ 
 				 (ПустоеЗначение(Наименование)=1) ИЛИ 
 				 ((ПустоеЗначение(ЦенаОптовая)=1) И (Акционный<>0)) ИЛИ
 				 ((ПустоеЗначение(ЦенаРозничная)=1) И (Акционный<>0)) ИЛИ
 				 ((ПустоеЗначение(ЦенаПервойКатегории)=1) И (Акционный<>0)) ИЛИ
 				 ((ПустоеЗначение(ЦенаВторойКатегории)=1) И (Акционный<>0)) ИЛИ
 				 ((ПустоеЗначение(ЦенаТретьейКатегории)=1) И (Акционный<>0)) ИЛИ
 				 ((ПустоеЗначение(Акционный)=1) И (Акционный<>0)) Тогда

 				Результат = 1/0;
 			КонецЕсли;	
			
 		Исключение
 			СтрокиСОшибками = СтрокиСОшибками + НомерСтроки + ", ";
 		КонецПопытки;
 	КонецЦикла;
	
	Если ПустоеЗначение(СтрокиСОшибками)=1 Тогда
		Сообщить("Файл без ошибок!"); 
		Сообщить("INFO: Всего строк в файле " + КолСтрок);
		Возврат(1);
	Иначе
		Сообщить("WARN: Файл с ошибками. Проблема в строках - " + СтрокиСОшибками);
		Возврат(0);
	КонецЕсли;
КонецФункции


Процедура ЗагрузитьДанные()
	// Блокируем кнопки во избежание прерывания апдейта
	Форма.кнИмпорт.Доступность(0);
	Форма.кнЗакрыть.Доступность(0);
	
	КорректностьФайла = ПроверитьФайлНаКорректность();
	
	Если КорректностьФайла=0 Тогда
		Возврат;
	КонецЕсли;
	
	СтрокВФайле = Excel.Cells.CurrentRegion.Rows.Count;
		
	Для СтрокаФайл = 2 По СтрокВФайле Цикл

		Код = СокрЛП(Excel.Cells(СтрокаФайл, 1).Value);
		Наименование = СокрЛП(Excel.Cells(СтрокаФайл, 2).Value);
		ЦенаОптовая = Число(Excel.Cells(СтрокаФайл, 3).Value);
		ЦенаРозничная = Число(Excel.Cells(СтрокаФайл, 4).Value);
		Цена1Кат = Число(Excel.Cells(СтрокаФайл, 5).Value);
		Цена2Кат = Число(Excel.Cells(СтрокаФайл, 6).Value);
		Цена3Кат = Число(Excel.Cells(СтрокаФайл, 7).Value);
		Акционный = Число(Excel.Cells(СтрокаФайл, 8).Value);
		АфмИд=Число(Excel.Cells(СтрокаФайл, 9).Value);
		КатИд=Число(Excel.Cells(СтрокаФайл, 10).Value);
		ДепИд=Число(Excel.Cells(СтрокаФайл, 11).Value);
		БрендИд=Число(Excel.Cells(СтрокаФайл, 12).Value);
		КачИд=Число(Excel.Cells(СтрокаФайл, 13).Value);
		СовмИд=Число(Excel.Cells(СтрокаФайл, 14).Value);
		
		
		СпрТМЦ = СоздатьОбъект("Справочник.ТМЦ");
		
		
		Если СпрТМЦ.НайтиПоКоду(Код)=1 Тогда
			СпрТМЦ.Наименование = Наименование;
			СпрТМЦ.ЦенаОптовая.Установить(ТекущаяДата(),ЦенаОптовая);
			СпрТМЦ.ЦенаРозничная.Установить(ТекущаяДата(),ЦенаРозничная);
			СпрТМЦ.Цена1Кат.Установить(ТекущаяДата(),Цена1Кат);
			СпрТМЦ.Цена2Кат.Установить(ТекущаяДата(),Цена2Кат);
			СпрТМЦ.Цена3Кат.Установить(ТекущаяДата(),Цена3Кат);
			СпрТМЦ.Акционный = Акционный;
			СпрТМЦ.АфмИд = АфмИд;
			СпрТМЦ.КатИд = КатИд;
			СпрТМЦ.ДепИд = ДепИд;
			СпрТМЦ.БрендИд = БрендИд;
			СпрТМЦ.КачИд = КачИд;
			СпрТМЦ.СовмИд = СовмИд;

  		
		 	СпрТМЦ.Записать();
		 	Состояние("Обработано строк " + (СтрокаФайл-1) + " из " + СтрокВФайле);
		
		Иначе
		    Сообщить("INFO: Товар с кодом " + Код + " не найден");
		КонецЕсли;
		

	КонецЦикла;	
	
	Сообщить("INFO: Импорт завершен успешно!");
	Форма.кнЗакрыть.Доступность(1);
	           
КонецПроцедуры
