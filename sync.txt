Функция ПолучитьХэш(ИсходныеДанные, Хэш=5381, М=33, Разрядность=18446744073709551616)

    СтрокаДляКодирования = Строка(ИсходныеДанные);

    ДлинаБлока = 11;
    НачПозиция = 1;
    ДлинаСтроки = СтрДлина(СтрокаДляКодирования);

    Пока НачПозиция <= ДлинаСтроки Цикл
        СтрокаБлока = Сред(СтрокаДляКодирования, НачПозиция, ДлинаБлока);
        ДлинаПодстроки = СтрДлина(СтрокаБлока);
        Если ДлинаПодстроки = ДлинаБлока Тогда
            Хэш = ((((((((((( Хэш*М + КодСимв(Сред(СтрокаБлока, 1, 1)))*М + КодСимв(Сред(СтрокаБлока, 2, 1)))*М
                + КодСимв(Сред(СтрокаБлока, 3, 1)))*М + КодСимв(Сред(СтрокаБлока, 4, 1)))*М + КодСимв(Сред(СтрокаБлока, 5, 1)))*М
                + КодСимв(Сред(СтрокаБлока, 6, 1)))*М + КодСимв(Сред(СтрокаБлока, 7, 1)))*М + КодСимв(Сред(СтрокаБлока, 8, 1)))*М
                + КодСимв(Сред(СтрокаБлока, 9, 1)))*М + КодСимв(Сред(СтрокаБлока, 10, 1)))*М + КодСимв(Сред(СтрокаБлока, 11, 1)))
        Иначе
            Для к = 1 По ДлинаПодстроки Цикл
                Хэш = М * Хэш + КодСимв(Сред(СтрокаБлока, к, 1))
            КонецЦикла
        КонецЕсли;
        Хэш = Хэш % Разрядность;
        НачПозиция = НачПозиция + ДлинаБлока
    КонецЦикла;

    Возврат Хэш;

Функция ПреобразоватьСтрокуВУТФ8(Стр)
    Длина=СтрДлина(Стр);
    Итог="";
    Для Н=1 По Длина Цикл
        Знак=Сред(Стр,Н,1);
        Код=КодСимв(Знак);
        Если Код<128 Тогда
            Итог=Итог+Знак;
        Иначе
            Если (Код>=КодСимв("А"))И(Код<=КодСимв("п")) Тогда
                Итог=Итог+Симв(208)+Симв(144+Код-КодСимв("А"));
            ИначеЕсли (Код>=КодСимв("р"))И(Код<=КодСимв("я")) Тогда
                Итог=Итог+Симв(209)+Симв(128+Код-КодСимв("р"));
            ИначеЕсли (Знак="ё") Тогда
                Итог=Итог+Симв(209)+Симв(145);
            ИначеЕсли (Знак="Ё") Тогда
                Итог=Итог+Симв(208)+Симв(129);
			ИначеЕсли (Знак = "і") Тогда
                Итог = Итог + Симв(209) + Симв(150);
            ИначеЕсли (Знак = "І") Тогда
                Итог = Итог + Симв(208) + Симв(134);
            ИначеЕсли (Знак = "ї") Тогда
                Итог = Итог + Симв(209) + Симв(151);
            ИначеЕсли (Знак = "Ї") Тогда
                Итог = Итог + Симв(208) + Симв(135);
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
    Возврат Итог;

Функция СтрокаJson(Параметр, Значение, ТипЗн = 0, ПоследняяСтрока = 0) 
	ПараметрДляДобавления = Симв(34) + Параметр + Симв(34);
	ЗначениеДляДобавления = "";
	
	Если ТипЗн = 1 Тогда
		ЗначениеДляДобавления = Число(Значение);
	ИначеЕсли ТипЗн = 2 Тогда
		ЗначениеДляДобавления = "[" + Значение + "]";
	ИначеЕсли ТипЗн = 3 Тогда
		ЗначениеДляДобавления = СтрЗаменить(Значение,Симв(34), "");
	Иначе
		ФорматЗначение = ОчиститьТекстОтПереносов(Значение);
		
		Если Найти(ФорматЗначение, Симв(92)) <> 0 Тогда
			ЭкранированныйОбратныйСлеш = Симв(92) + Симв(92); //  \ -> \\ 
			ФорматЗначение = СтрЗаменить(ФорматЗначение,Симв(92), ЭкранированныйОбратныйСлеш); // экранируем обратный слеш для json			
		КонецЕсли;
		
		
		Если Найти(ФорматЗначение, Симв(34)) <> 0 Тогда
			ЭкранированнаяКавычка = Симв(92) + Симв(34); //  \" 
			ФорматЗначение = СтрЗаменить(ФорматЗначение,Симв(34), ЭкранированнаяКавычка); // экранируем кавычки для json			
		КонецЕсли;
		                                                         
		ЗначениеДляДобавления = Симв(34) + ФорматЗначение + Симв(34);
	КонецЕсли;
	
	Если ПоследняяСтрока = 1 Тогда
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления);
	Иначе
		Возврат(ПараметрДляДобавления + ":" + ЗначениеДляДобавления + ",");
	КонецЕсли;

Функция ОтправитьPOSTЗапрос(URL, ТекстJSON)
   
	ЗагрузитьВнешнююКомпоненту("OleExSup.dll");
	OleExSup = СоздатьОбъект("OLEExSup");
	
	ТелоОтветаНаЗапрос = СоздатьОбъект("СписокЗначений");

    Попытка                                                                                              
        HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
	Исключение
		ТелоОтветаНаЗапрос.ВставитьЗначение(1, 500);
		ТелоОтветаНаЗапрос.ВставитьЗначение(2, "Не удалось создать COM-объект WinHttpRequest! " + ОписаниеОшибки());
        Возврат ТелоОтветаНаЗапрос;
    КонецПопытки;

    // Открываем соединение: ("Метод", "URL", Асинхронный = 0)
    HTTP.Open("POST", URL, 0);

    // Устанавливаем заголовки. Content-Type обязателен для JSON.
    // Указание кодировки UTF-8.
    HTTP.SetRequestHeader("Content-Type", "application/json; charset=utf-8");

    // Отправляем тело запроса (нашу строку JSON)
    Попытка
        // HTTP.Send(ТекстJSON);
		OleExSup.InvokeOLEMethod(HTTP, "Send", ТекстJSON);
	Исключение
		ТелоОтветаНаЗапрос.ВставитьЗначение(1, 500);
		ТелоОтветаНаЗапрос.ВставитьЗначение(2, "Ошибка при отправке POST-запроса: " + ОписаниеОшибки());
        Возврат ТелоОтветаНаЗапрос;
    КонецПопытки;


    // Проверяем результат
    Если HTTP.Status = 200 Тогда // 200 - это стандартный код успешного ответа "OK"
        // Запрос прошел успешно, возвращаем текст ответа от сервера
		ОтветССервера = HTTP.ResponseText;
		Сообщение = "SERVER OK: Статус: " + HTTP.Status + ", Сообщение от сервера: " + ПолучЗначениеПараметраСJson("message",ОтветССервера);
		ТелоОтветаНаЗапрос.ВставитьЗначение(1, 200);
		ТелоОтветаНаЗапрос.ВставитьЗначение(2, Сообщение);
		
        Возврат ТелоОтветаНаЗапрос;
    Иначе
        // Если сервер вернул ошибку
		ОтветССервера = HTTP.ResponseText;
		Сообщение = "SERVER ERROR: Статус: " + HTTP.Status + ", Сообщение от сервера: " + ПолучЗначениеПараметраСJson("message",ОтветССервера);
		ТелоОтветаНаЗапрос.ВставитьЗначение(1, 500);
		ТелоОтветаНаЗапрос.ВставитьЗначение(2, Сообщение);
        Возврат ТелоОтветаНаЗапрос;
	КонецЕсли;

// получаем объект контрагента по ид, и цену конкретного товара для него (опционально)
Функция ПолучитьКонтрагентаИЦенуТовара(ИдКонтрагента, ИдТовара = "");
	
	СпрКонтрагенты = СоздатьОбъект("Справочник.Контрагенты");
	СпрТмц = СоздатьОбъект("Справочник.ТМЦ");
	КатегорияЦенКонтрагента = "Розничная";
	Ответ = СоздатьОбъект("СписокЗначений");
	Ответ.ВставитьЗначение(1, 0);
	Ответ.ВставитьЗначение(2, 0);
	Ответ.ВставитьЗначение(3, 0);

	Если СпрКонтрагенты.НайтиПоКоду(ИдКонтрагента) = 1 Тогда
		Контрагент = СпрКонтрагенты.ТекущийЭлемент();
		КатегорияЦенКонтрагента = Контрагент.КатегорияЦенКонтрагента;
		Ответ.ВставитьЗначение(1, Контрагент);                                      
	Иначе
		Возврат(Ответ); // возвращаем пустой массив, т.к. из-за отсутствия пользователя нельзя оперделить цену товара
	КонецЕсли;
	
	Если ПустоеЗначение(ИдТовара) = 1 Тогда
		Возврат(Ответ);  // Возвращаем только пользователя
	КонецЕсли;
	
	
	Если СпрТмц.НайтиПоКоду(ИдТовара) <> 1 Тогда
		Возврат(Ответ);
	КонецЕсли;
		
	Тмц = СпрТмц.ТекущийЭлемент();
	ТмцЦена = 0;
	
	Если СокрЛП(КатегорияЦенКонтрагента) = "Цена 1 категории" Тогда
		ТмцЦена = Тмц.Цена1Кат.Получить(ТекущаяДата());
	ИначеЕсли СокрЛП(КатегорияЦенКонтрагента) = "Цена 2 категории" Тогда
		ТмцЦена = Тмц.Цена2Кат.Получить(ТекущаяДата());
	ИначеЕсли СокрЛП(КатегорияЦенКонтрагента) = "Цена 3 категории" Тогда
		ТмцЦена = Тмц.Цена3Кат.Получить(ТекущаяДата());
	ИначеЕсли  СокрЛП(КатегорияЦенКонтрагента) = "Оптовая" Тогда
		ТмцЦена = Тмц.ЦенаОптовая.Получить(ТекущаяДата())
	Иначе
		Сообщить("Тмц.ЦенаРозничная");
		ТмцЦена = Тмц.ЦенаРозничная.Получить(ТекущаяДата());
	КонецЕсли;

    Ответ.ВставитьЗначение(2, ТмцЦена);
	Ответ.ВставитьЗначение(3, Тмц);
	Возврат(Ответ);	
		   
КонецФункции

Функция ПолучитьОснованиеЗаказаРН(Заказ)
	ЗапросРН = СоздатьОбъект("Запрос");

	ТекстЗапрос = "
	    |Период с НачДатаСинхронизации по КонДатаСинхронизации;
		|Обрабатывать НеПомеченныеНаУдаление;
	    |Док = Документ.РасходнаяНакладная.ТекущийДокумент;
	    |ДокКод = Документ.РасходнаяНакладная.НомерДок;
		|ДокОснование = Документ.РасходнаяНакладная.ДокОснование;                     
		|Условие (ДокОснование.НомерДок = Заказ.ДокКод);
		|Условие (ДокОснование.ДатаДок = Заказ.ДокДата);
		|Группировка Док;
	    |"; 
		
	Если ЗапросРН.Выполнить(ТекстЗапрос) = 0 Тогда
		Сообщить("Запрос по поиску РН для заказа не выполнен.");
		Возврат("");
	КонецЕсли;
	
	Пока ЗапросРН.Группировка("Док") = 1 Цикл
		Возврат(ЗапросРН.Док);
	КонецЦикла;
	  
	Возврат(0);

Функция ОтправитьОбработанныеЗаказыНаСервер();
	ЗапросЗаказы = СоздатьОбъект("Запрос");
	ЗаказыJson = "";
	
	ТекстЗапрос = "
	    |Период с НачДатаСинхронизации по КонДатаСинхронизации;
		|Обрабатывать НеПомеченныеНаУдаление;
	    |ОбрабатыватьДокументы НеПроведенные;
	    |Док = Документ.ЗаказПокупателя.ТекущийДокумент;
	    |ДокКод = Документ.ЗаказПокупателя.НомерДок;
		|ДокРн = Документ.ЗаказПокупателя.ДокРН;
		|ДокДата = Документ.ЗаказПокупателя.ДатаДок;
	    |ДокКодНаСайте = Документ.ЗаказПокупателя.НомерЗаказаНаСайте;
	    |ДокПримечание = Документ.ЗаказПокупателя.Примечание;
		|ДокОбработан = Документ.ЗаказПокупателя.Готов;
		|ДокТмцСумма = Документ.ЗаказПокупателя.Сумма;
		|Функция ДокСумма = Сумма(ДокТмцСумма);
	    |Группировка Док;
		|Условие (ДокОбработан = 1);
	    |";
	
	Если ЗапросЗаказы.Выполнить(ТекстЗапрос) = 0 Тогда
		Сообщить("Выполнить() вернул 0 — запрос не вернул строк. Проверь структуру запроса и доступность полей.");
		Возврат(0);
	КонецЕсли;
	                                           
	Пока ЗапросЗаказы.Группировка("Док") = 1 Цикл
		РасхНакл = ПолучитьОснованиеЗаказаРН(ЗапросЗаказы);
		
		// Если не создана еще РН, то будет пропущен
		Если ПустоеЗначение(РасхНакл) = 1 Тогда
			Продолжить;	
		КонецЕсли;
		РасходнаяНакладнаяХэш = ПолучитьХэш(РасхНакл.НомерДок+РасхНакл.ДатаДок);
		
		ЗаказНаСайтеКод = ЗапросЗаказы.ДокКодНаСайте;
		ЗаказКод = ЗапросЗаказы.ДокКод;
		Обработано = ЗапросЗаказы.ДокОбработан;
		ЗаказСумма = ЗапросЗаказы.ДокСумма;
		ЗаказПримечание = ЗапросЗаказы.ДокПримечание;
		ЗаказДата = ЗапросЗаказы.ДокДата;
		ЗаказХэш = ПолучитьХэш(ЗаказКод+ЗаказДата);
		
		ЗаказJson = "{"
			+ СтрокаJson("id", ЗаказНаСайтеКод,1,0)
			+ СтрокаJson("id1c", ЗаказКод,,0)
			+ СтрокаJson("orderHash", ЗаказХэш,,0)
			+ СтрокаJson("processed", ?(Обработано = 1, "true", "false"),3,0)
			+ СтрокаJson("invoiceId", РасхНакл.НомерДок,,0)
			+ СтрокаJson("invoiceHash", РасходнаяНакладнаяХэш,,0)
			+ СтрокаJson("totalAmount", ЗаказСумма,1,0)
			+ СтрокаJson("comment",ЗаказПримечание,,1)
			+ "}";
		
		ЗаказыJson = ЗаказыJson + ЗаказJson + ",";
	КонецЦикла;
	
	ЗаказыJson = "[" + УдалитьПоследнююЗапятую(СокрЛП(ЗаказыJson)) + "]";
	
	Запрос = ОтправитьPOSTЗапрос(ЗапросАдресСервера + "/orders", ЗаказыJson);
	
	Если Запрос.ПолучитьЗначение(1) <> 200 Тогда
		Сообщить(Запрос.ПолучитьЗначение(2));
		Сообщить("ERR: Ошибка при сихронизации обработанных заказов.");
	Иначе
		Сообщить(Запрос.ПолучитьЗначение(2));
	КонецЕсли;      
	Возврат(0);
КонецФункции


Процедура ПолучитьЗаказыССервера()
	// Файл для сохранения с сервера/чтения CSV данных
	ПутьКФайлу = РабочаяПапка + "ordersFromServer.txt"; 
	Запрос = ОтправитьGETЗапрос(ЗапросАдресСервера + "/orders");
	Статус = Запрос.ПолучитьЗначение(1);
	Сообщение = Запрос.ПолучитьЗначение(2);
	
	// запрашиваем новые заказы с сервера
	Если Статус <> 200 Тогда		
		Сообщить("ERR: Ошибка при получении заказов с сервера. Синхронизация не завершена");
		Возврат;
	Иначе
		ДанныеССервера = Запрос.ПолучитьЗначение(3);
		ФорматДанныеССервера = СтрЗаменить(ДанныеССервера, "^", РазделительСтрок); // делаем перенос строк как в СSV
		ЗаписьТекстовогоФайла(РабочаяПапка + "ordersFromServer.txt",ФорматДанныеССервера);
		Сообщить(Сообщение);		
		Сообщить("OK: Заказы успешно получены с сервера и записаны в файл. ");
	КонецЕсли;
	            
	//Открываем записаный файл с заказами
	ЗаказыССервера = ОткрытьТекстовыйФайл(РабочаяПапка + "ordersFromServer.txt");
	
	// Создаем таблицу для хранения данных из файла
	ТЗ = СоздатьОбъект("ТаблицаЗначений");
	ТЗ.НоваяКолонка("КодПользователя", "Строка", 20);
    ТЗ.НоваяКолонка("КодЗаказаССайта", "Строка", 20);
    ТЗ.НоваяКолонка("КодТовара", "Строка", 20);
    ТЗ.НоваяКолонка("Кво", "Число", 10, 3);
    ТЗ.НоваяКолонка("Комментарий", "Строка", 255);

	// таблица значений
	// обработка файла по-строчно. Начинаем со 2й строки, т.к. в 1й строке - шапка
	Для Ном = 2 По ЗаказыССервера.КоличествоСтрок() Цикл
		СтрЗаказа = ЗаказыССервера.ПолучитьСтроку(Ном);
		
		Если ПустаяСтрока(СтрЗаказа) = 1 Тогда
			Продолжить;
		КонецЕсли;
		
		// Парсим строку для получения значений
		ЗначСтр = РазложитьСтрокуВМассив(СтрЗаказа);
		
		ТЗ.НоваяСтрока();
		
		ТЗ.КодПользователя = СокрЛП(ЗначСтр.ПолучитьЗначение(1));
		ТЗ.КодЗаказаССайта = СокрЛП(ЗначСтр.ПолучитьЗначение(2));
		ТЗ.КодТовара 	   = СокрЛП(ЗначСтр.ПолучитьЗначение(3));
		ТЗ.Кво 			   = Число(СокрЛП(ЗначСтр.ПолучитьЗначение(4)));
		ТЗ.Комментарий 	   = СокрЛП(ЗначСтр.ПолучитьЗначение(5));
	КонецЦикла;
	
	Если ТЗ.КоличествоСтрок() = 0 Тогда
        Сообщить("Новых заказов нет", "!");
        Возврат;
	КонецЕсли;
	
	// Сортируем по коду заказа, чтобы строки товаров одного заказа шли друг за другом
	ТЗ.Сортировать("КодЗаказаССайта");
	
	
	// Создание Документов
	СпрСклады = СоздатьОбъект("Справочник.Склады");
	СпрВалюты = СоздатьОбъект("Справочник.Валюты");
	
	Док = 0;
	ТекущийКодЗаказаССайта = "";
	Склад = Константа.ОсновнойСклад;
	Валюта = Константа.ОсновнаяВалюта;
	
	ТЗ.ВыбратьСтроки();
	Пока ТЗ.ПолучитьСтроку() = 1 Цикл
					
		// проверка на существование заказа в 1с, по ид сайта
		ЗапросТекЗаказы = СоздатьОбъект("Запрос"); 
		
		ТекстЗапрос = "
			|Обрабатывать НеПомеченныеНаУдаление;
		    |ОбрабатыватьДокументы НеПроведенные;
		    |Док = Документ.ЗаказПокупателя.ТекущийДокумент;
		    |ДокКодНаСайте = Документ.ЗаказПокупателя.НомерЗаказаНаСайте;
			|ДокОбработан = Документ.ЗаказПокупателя.Готов; 
		    |Группировка Док;
			|Условие (СокрЛП(ДокКодНаСайте) = СокрЛП(ТЗ.КодЗаказаССайта));
		    |";
			                                                                      
		ЗапросТекЗаказы.Выполнить(ТекстЗапрос);
		
		ЗаказУжеЕстьВБазе = 0;
		
		Пока ЗапросТекЗаказы.Группировка("Док") = 1 Цикл
			// если заказ уже создан в бд, пропускаем
			ЗаказУжеЕстьВБазе = 1;
			Продолжить;
		КонецЦикла;
		
		Если ЗаказУжеЕстьВБазе = 1 Тогда
			Продолжить;
		КонецЕсли;
		
		// Если изменился номер заказа - создаем новый док-т
		Если  ТекущийКодЗаказаССайта <> ТЗ.КодЗаказаССайта Тогда			
			// Сохраняем предыдущий документ, если он был
			Если Док <> 0 Тогда
				Попытка             
					Док.Записать();
				Исключение
					Сообщить("Ошибка при записи документа, для заказа " + ТекущийКодЗаказаССайта);
				КонецПопытки;
			КонецЕсли;
 
			Док = СоздатьОбъект("Документ.ЗаказПокупателя");
			Док.Новый();
			Док.ДатаДок = ТекущаяДата();
			
			
			
			
			//---- получаем контрагента для добавления в шапку
			
			Контрагент = ПолучитьКонтрагентаИЦенуТовара(ТЗ.КодПользователя);
			
			Если ПустоеЗначение(Контрагент.ПолучитьЗначение(1)) = 1 Тогда
				Сообщить("Пользователь с ИД "+ТЗ.КодПользователя+" не найден");
				Продолжить;
			КонецЕсли;
			
			Док.Клиент = Контрагент.ПолучитьЗначение(1);
			
			//----
			
		    Док.Примечание = ТЗ.Комментарий;
			Док.МестоХранения = Склад;
			Док.Валюта = Валюта;
			Док.НомерЗаказаНаСайте = ТЗ.КодЗаказаССайта;
			ТекущийКодЗаказаССайта = ТЗ.КодЗаказаССайта;
		КонецЕсли;
		
		// Добавляем товары
		
		Товар = ПолучитьКонтрагентаИЦенуТовара(ТЗ.КодПользователя, ТЗ.КодТовара);
		
		Если ПустоеЗначение(Товар.ПолучитьЗначение(3)) = 1 Тогда
			Сообщить("Товар с ИД " + ТЗ.КодТовара + " не наден, и будет пропущен в заказе");
		КонецЕсли;
			
		Док.НоваяСтрока();
		Док.ТМЦ = Товар.ПолучитьЗначение(3);
		Док.Кво = ТЗ.Кво;
		Док.Цена = Товар.ПолучитьЗначение(2);
		Док.Сумма = Окр(ТЗ.Кво * Товар.ПолучитьЗначение(2), 2);
	КонецЦикла;
	
	// Сохраняем последний документ
	
	Если Док <> 0 Тогда
		Попытка
			Док.Записать();
		Исключение
			Сообщить("Ошибка при записи документа, для заказа " + ТекущийКодЗаказаССайта);
			Возврат;
		КонецПопытки;
	КонецЕсли;	
	
КонецПроцедуры
